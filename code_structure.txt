### 詳細な実装の構成
pyquant/
├── backend/                # FastAPI アプリケーション
│   ├── app/
│   │   ├── api/            # APIエンドポイント定義（取引登録、評価実行など）
│   │   ├── core/           # 設定（DB接続設定、共通定数など）
│   │   ├── db/             # SQLAlchemy Model, DB接続管理, Migration
│   │   ├── repository/     # データアクセス層 (CRUD操作)
│   │   └── engine/         # 【クオンツ核】数学・金融工学ロジック
│   │       ├── math/       # 補間, 複利計算, 統計, 数値計算 (Numpy/Scipy)
│   │       ├── models/     # 価格モデル (Black, Garman-Kohlhagen, etc.)
│   │       ├── products/   # 商品別ロジック (IRS, Bond, FX, etc.)
│   │       ├── market/     # 曲線構築 (Bootstrap), ボラ面構築
│   │       └── kernels/    # 高速化ロジック (Numba, Cupy用の実装を分離)
│   │   ├── schemas/        # API通信用の型定義（Pydantic）
│   │   └── services/       # 業務ロジック（マーケット版の確定、Fixing解決等）
│   ├── tests/              # ユニットテスト（特に数理ロジックの検証用）
│   ├── main.py             # アプリケーションのエントリポイント
│   └── requirements.txt    # 依存ライブラリ
├── frontend/               # React アプリケーション
│   ├── src/
│   │   ├── api/            # バックエンドAPIを呼び出す関数群
│   │   ├── components/     # 再利用可能なUIパーツ（テーブル、ダイアログ等）
│   │   ├── hooks/          # カスタムフック（状態管理、API通信）
│   │   ├── pages/          # 各画面（取引管理、マーケット管理、評価実行等）
│   │   ├── types/          # TypeScript型定義
│   │   └── utils/          # フォーマッタ（日付、金額の表示形式変換）
│   └── package.json
├── data/                   # SQLite データベースファイル保存先
├── docs/                   # 設計ドキュメント
└── scripts/                # データ取込バッチや初期化用スクリプト
